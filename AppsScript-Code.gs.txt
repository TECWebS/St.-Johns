function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const sheet = getOrCreateSheet_(data.formType);
    writeRow_(sheet, data);
    return ContentService
      .createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput("St. John's Christian Church form endpoint is live.");
}

function getOrCreateSheet_(formType) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tabName = formType === "connectCard" ? "Connect Cards" : "Contact Messages";
  let sheet = ss.getSheetByName(tabName);
  if (!sheet) {
    sheet = ss.insertSheet(tabName);
    const headers = formType === "connectCard"
      ? ["Submitted At", "Name", "Email", "Phone", "Interests", "Message"]
      : ["Submitted At", "Name", "Email", "Message"];
    sheet.appendRow(headers);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function writeRow_(sheet, data) {
  const now = new Date();
  if (data.formType === "connectCard") {
    sheet.appendRow([
      now,
      data.name || "",
      data.email || "",
      data.phone || "",
      (data.interests || []).join(", "),
      data.message || "",
    ]);
  } else {
    sheet.appendRow([
      now,
      data.name || "",
      data.email || "",
      data.message || "",
    ]);
  }
}
